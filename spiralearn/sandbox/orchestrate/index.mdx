# StudyLenses — checkpoint

The full instrument, live, with the scaffolding level, the three ported lenses
(parsons and writeme in the Source select; debug-props is panel-excluded —
reachable only as a focus request), and one page-injected demonstration lens
("notes", named as the initial focus). Expected: the editor mounts with the seed
program and the notes lens opens at load, honored, without a click; typing until
the debounce settles re-renders the phases; breaking the parse bars the
downstream phases with the parser's cause and fixing it reopens them; the type
toggle re-derives immediately; the level selector shows the scaffold's live fit
mark; under strict with a `debugger;` the masked overlay covers the panel AND
the notes lens identically; the guide reveals its topics after the panel.

export function InstrumentHarness() {
	const React = require('react');
	const StudyLenses =
		require('@site/src/lib/study-lenses/orchestrate/index.tsx').default;
	const scaffoldLevel =
		require('@site/src/lib/study-lenses/language-levels/scaffold/index.ts').default;
	const parsonsLens =
		require('@site/src/lib/study-lenses/lenses/parsons/index.tsx').default;
	const writemeLens =
		require('@site/src/lib/study-lenses/lenses/writeme/index.tsx').default;
	const debugPropsLens =
		require('@site/src/lib/study-lenses/lenses/debug-props/index.tsx').default;
	const notesLens = {
		name: 'notes',
		applicability: () => true,
		phase: 'source',
		main: ({ embodiment, config }) => (
			<p data-notes-lens>
				notes (opened via {String(config.origin ?? 'the panel')}) — this program
				is {embodiment.facts.source.value.length} characters of source
			</p>
		),
	};
	notesLens.recommend = () => [
		{
			lens: notesLens,
			config: { origin: 'a recommendation' },
			relevance: 0.8,
			label: 'study the source',
		},
	];
	return (
		<StudyLenses
			languageLevels={[scaffoldLevel]}
			lens="notes"
			lenses={[notesLens, parsonsLens, writemeLens, debugPropsLens]}
			snippet={'const greeting = "hello";\nconsole.log(greeting);\n'}
		/>
	);
}

<BrowserOnly fallback={<p>loading the instrument…</p>}>
	{() => <InstrumentHarness />}
</BrowserOnly>
